**Step B — module skeleton.** Create `crates/holt-cli/src/install_hooks/mod.rs` (CR-style block):

```rust
//! `holt install-hooks` library — JSONC-aware merge of holt's hook entries
//! into `~/.claude/settings.json`.
//!
//! Phase 3 hard constraints:
//!   - C3: fs2 exclusive lock + fsync-before-rename + PID-suffix tmp + `.holt.bak` backup.
//!   - C4: jsonc-parser dep lives ONLY in this crate (verified by
//!         `tests/cli_dep_boundary.rs` at workspace root in plan 03-02).
//!
//! Public surface consumed by `holt-cli/src/install_hooks_cmd.rs` (plan 03-02):
//!   - `merge_settings(input: &str) -> Result<MergeOutput, MergeError>`
//!   - `commit(home: &Path, merged: &MergeOutput) -> Result<(), CommitError>`
//!   - `entries::HOLT_HOOK_ENTRIES`
//!
//! Single-backup policy (D-07): `~/.claude/settings.json.holt.bak` is overwritten
//! on every successful run. NOT a versioned backup chain — long-lived user files
//! shouldn't accumulate cruft. NOT named `.bak` (vim's namespace per
//! research/SUMMARY.md §3 C3).

#![allow(dead_code)] // wired into the binary in plan 03-02

mod entries;
mod lock;
mod merge;

pub use entries::{HOLT_HOOK_ENTRIES, HoltHookEntry};
pub use lock::{LockError, acquire_settings_lock};
pub use merge::{MergeError, MergeOutput, merge_settings};

use std::path::Path;

/// Errors at the commit stage (after merge succeeded, before the .bak + atomic_write).
#[derive(Debug, thiserror::Error)]
pub enum CommitError {
    #[error("failed to write .holt.bak backup at {path}: {source}")]
    BackupWrite { path: String, source: std::io::Error },
    #[error("failed to atomically write merged settings.json at {path}: {source}")]
    AtomicWrite { path: String, source: std::io::Error },
}

/// Backup pre-merge bytes to `<settings_path>.holt.bak`, then atomically write merged bytes
/// to `<settings_path>` via `holt_schemas::atomic_write` (D-06 reuses Phase 1's PID-suffix
/// tmp + fsync-before-rename helper unchanged).
pub fn commit(
    settings_path: &Path,
    pre_merge: &[u8],
    merged: &MergeOutput,
) -> Result<(), CommitError> {
    let bak_path = {
        let mut s = settings_path.as_os_str().to_owned();
        s.push(".holt.bak");
        std::path::PathBuf::from(s)
    };
    holt_schemas::atomic_write(&bak_path, pre_merge).map_err(|e| CommitError::BackupWrite {
        path: bak_path.display().to_string(),
        source: e,
    })?;
    holt_schemas::atomic_write(settings_path, merged.bytes.as_bytes()).map_err(|e| {
        CommitError::AtomicWrite {
            path: settings_path.display().to_string(),
            source: e,
        }
    })?;
    Ok(())
}
```

```rust
//! D-04 fs2 exclusive lock with 200ms try-loop.
//!
//! Pattern: try_lock_exclusive once; if it returns ErrorKind::WouldBlock,
//! sleep 50ms and try again, up to 4 attempts total (200ms budget). On the
//! fourth failure, return LockError::Timeout with a user-facing error message
//! that names both "holt install-hooks" and "settings.json" so support
//! tickets can grep for either keyword.
//!
//! On success, returns the open File handle. The caller MUST keep the handle
//! alive for the entire read-merge-write window; the lock is released on Drop.
//!
//! C3 mandate: lock is acquired on the settings.json file itself, NOT a
//! separate lock file. fs2's exclusive lock is advisory but every other
//! `holt install-hooks` invocation respects it (POSIX flock(2) / Windows
//! LockFileEx).

use fs2::FileExt;
use std::fs::OpenOptions;
use std::io::ErrorKind;
use std::path::Path;
use std::thread;
use std::time::Duration;

const MAX_ATTEMPTS: u32 = 4;
const SLEEP_BETWEEN_ATTEMPTS: Duration = Duration::from_millis(50);
const TOTAL_BUDGET_MS: u64 = 200;

#[derive(Debug, thiserror::Error)]
pub enum LockError {
    #[error(
        "could not acquire exclusive lock on {path} after {budget_ms}ms: another holt install-hooks is running (or settings.json is locked by another editor)"
    )]
    Timeout { path: String, budget_ms: u64 },
    #[error("io error opening {path} for locking: {source}")]
    Io {
        path: String,
        #[source]
        source: std::io::Error,
    },
}

/// Open `path` (creating it as 0o600 with `{}` content if absent), acquire
/// an exclusive fs2 lock with 200ms budget, return the locked File.
pub fn acquire_settings_lock(path: &Path) -> Result<std::fs::File, LockError> {
    // Ensure parent directory exists (e.g., ~/.claude/ on a fresh system).
    if let Some(parent) = path.parent() {
        if !parent.exists() {
            std::fs::create_dir_all(parent).map_err(|e| LockError::Io {
                path: parent.display().to_string(),
                source: e,
            })?;
        }
    }

    let mut opts = OpenOptions::new();
    opts.read(true).write(true).create(true);
    #[cfg(unix)]
    {
        use std::os::unix::fs::OpenOptionsExt;
        opts.mode(0o600);
    }

    // create(true) does NOT truncate; if the file exists with content, open it as-is.
    // If we just created it (0 bytes), seed with `{}` so JSONC parse succeeds at the
    // merge step. We handle the seeding off the lock path in the caller, NOT here —
    // this function's contract is "return a locked handle". The caller reads the
    // bytes; if empty, treats as `{}` (the "clean" fixture starting state).
    let f = opts.open(path).map_err(|e| LockError::Io {
        path: path.display().to_string(),
        source: e,
    })?;

    for attempt in 0..MAX_ATTEMPTS {
        match f.try_lock_exclusive() {
            Ok(()) => return Ok(f),
            Err(ref e) if e.kind() == ErrorKind::WouldBlock => {
                if attempt + 1 < MAX_ATTEMPTS {
                    thread::sleep(SLEEP_BETWEEN_ATTEMPTS);
                }
                continue;
            }
            Err(e) => {
                return Err(LockError::Io {
                    path: path.display().to_string(),
                    source: e,
                });
            }
        }
    }

    Err(LockError::Timeout {
        path: path.display().to_string(),
        budget_ms: TOTAL_BUDGET_MS,
    })
}
```

```rust
//! D-02: pure jsonc-parser CST round-trip. NO json_comments strip-then-parse.
//!
//! Strategy:
//!   1. Parse the input as CST (preserves comments + key order + trailing commas).
//!   2. Walk to the top-level `hooks` object (create if absent — appended AFTER existing keys).
//!   3. For each event in HOLT_HOOK_ENTRIES:
//!      a. If `hooks.<event>` is absent, create a new array containing holt's canonical block.
//!      b. If present, scan the array for any element whose `hooks[].command` contains
//!         HOLT_HOOK_DETECTION_SUBSTR ("holt hook "). If found, replace it in-place (D-10).
//!         If not found, append holt's canonical block (D-08 co-existence).
//!   4. Emit the CST as bytes. Return MergeOutput { bytes, changed }.
//!
//! `changed` is true if the post-merge bytes differ from the input. Idempotency invariant
//! (D-08): re-merging the output yields `changed == false`. This is the testable contract.

use jsonc_parser::cst::{
    CstContainerNode, CstInputValue, CstObject, CstObjectProp, CstRootNode,
};
use jsonc_parser::ParseOptions;

use super::entries::{HOLT_HOOK_DETECTION_SUBSTR, HOLT_HOOK_ENTRIES, HoltHookEntry};

#[derive(Debug, thiserror::Error)]
pub enum MergeError {
    #[error("settings.json is not valid JSONC: {0}")]
    Parse(String),
    #[error("settings.json root is not a JSON object (got {got})")]
    NotAnObject { got: &'static str },
}

pub struct MergeOutput {
    pub bytes: String,
    pub changed: bool,
}

pub fn merge_settings(input: &str) -> Result<MergeOutput, MergeError> {
    // Empty file = treat as `{}` (newly-created settings.json from lock acquire).
    let normalized = if input.trim().is_empty() { "{}" } else { input };

    let opts = ParseOptions {
        allow_comments: true,
        allow_trailing_commas: true,
        allow_loose_object_property_names: false,
    };
    let root = CstRootNode::parse(normalized, &opts)
        .map_err(|e| MergeError::Parse(e.to_string()))?;

    // CstRootNode → CstObject for the top-level value.
    let root_obj = root
        .object_value()
        .ok_or(MergeError::NotAnObject { got: "non-object root" })?;

    // Walk-or-create `hooks` object.
    let hooks_obj = match root_obj.object_value_by_name("hooks") {
        Some(v) => v,
        None => {
            // Append a new `hooks: {}` property AFTER existing keys (preserves key order).
            let new_prop = root_obj.append("hooks", CstInputValue::Object(vec![]));
            new_prop
                .value()
                .and_then(|v| v.into_object())
                .expect("just-appended hooks value is an object")
        }
    };

    // For each event: ensure exactly one holt entry exists in hooks.<event>.
    for entry in HOLT_HOOK_ENTRIES {
        upsert_event(&hooks_obj, entry);
    }

    let bytes = root.to_string();
    let changed = bytes != input;
    Ok(MergeOutput { bytes, changed })
}

fn upsert_event(hooks_obj: &CstObject, entry: &HoltHookEntry) {
    use jsonc_parser::cst::CstInputValue::*;

    // Build the canonical holt entry block (D-09 exact bytes):
    //   { "matcher": "*", "hooks": [ { "type": "command", "command": "holt hook <Event>" } ] }
    let canonical = || {
        Object(vec![
            ("matcher", String("*".into())),
            (
                "hooks",
                Array(vec![
                    Object(vec![
                        ("type", String("command".into())),
                        ("command", String(entry.command.into())),
                    ]),
                ]),
            ),
        ])
    };

    match hooks_obj.array_value_by_name(entry.event) {
        None => {
            // No existing array — create with one element (holt's).
            hooks_obj.append(entry.event, Array(vec![canonical()]));
        }
        Some(arr) => {
            // Scan for an existing holt entry by D-10 substring detection.
            let mut replaced = false;
            for (idx, elem) in arr.elements().into_iter().enumerate() {
                if element_command_contains(&elem, HOLT_HOOK_DETECTION_SUBSTR) {
                    arr.replace(idx, canonical());
                    replaced = true;
                    break;
                }
            }
            if !replaced {
                // D-08: co-exist, do not clobber. Append after user's entries.
                arr.append(canonical());
            }
        }
    }
}

/// True if `elem` is an object with a `hooks` array whose first element's
/// `command` field is a string containing `needle`. Anything else returns false
/// (defensive — user-malformed entries are NOT mistakenly classified as ours).
fn element_command_contains(elem: &CstContainerNode, needle: &str) -> bool {
    let Some(obj) = elem.as_object() else { return false; };
    let Some(hooks_arr) = obj.array_value_by_name("hooks") else { return false; };
    for inner in hooks_arr.elements() {
        let Some(inner_obj) = inner.as_object() else { continue; };
        let Some(cmd) = inner_obj.string_value_by_name("command") else { continue; };
        if cmd.contains(needle) {
            return true;
        }
    }
    false
}
```
